API 接入指南 （站长版·用户不用看）

约 20 分钟 · DeepSeek 账号 + Cloudflare 账号（都免费）

收费模式：每天免费 5 次（按 IP，引流用）+ 卡密付费（闲鱼手动发货，不用企业资质）。Key 藏在 Worker 里，用户只填卡密。

第一步：拿 DeepSeek API Key

  1. 打开 platform.deepseek.com 注册，左侧 API Keys 创建 Key（只显示一次，保存好）。
  2. 用 deepseek-v4-flash 模型（旧的 deepseek-chat 已退役）。

第二步：建 KV（存卡密和计数）

  1. Cloudflare 后台 → 左侧 Workers 和 Pages → KV → 创建命名空间，起名 xskai-codes。
  2. 创建 Worker（起名如 xskai-api）→ 设置 → 绑定 → 添加 KV 命名空间绑定，变量名填 CODES，选刚建的 xskai-codes。
  3. 同样在设置 → 变量里加三个文本变量：ADMIN_KEY（生成卡密的密码，自己定个复杂的）；MATHPIX_APP_ID 和 MATHPIX_APP_KEY（见下）。

第三步：注册 Mathpix（公式识别）

  1. 打开 mathpix.com 注册开发者账号，拿到 app_id 和 app_key，填进 Worker 变量（上一步）。
  2. Mathpix 是公式 OCR 的金标准，手写公式也能识别；注册有免费额度，超出后按量付费（美元）。
  3. 如果暂时不想接：不填也行，网站会自动降级用免费 OCR（公式识别效果差一些，纯文字没问题）。

第四步：部署 Worker 代码

把下面代码粘贴进 Worker，替换 你的_DEEPSEEK_KEY，点部署：

const API_KEY = "你的_DEEPSEEK_KEY";
const FREE_DAILY = 5; // 每天免费次数（按IP）

function cors(){return {"Access-Control-Allow-Origin":"*","Access-Control-Allow-Methods":"POST, OPTIONS","Access-Control-Allow-Headers":"Content-Type, X-Access-Code"};}
function err(msg, status){return new Response(JSON.stringify({error:msg}),{status:status,headers:{...cors(),"Content-Type":"application/json"}});}

export default {
async fetch(req, env){
  const url = new URL(req.url);
  if(req.method==="OPTIONS") return new Response(null,{headers:cors()});

  // --- 管理员：生成卡密 ---
  if(url.pathname==="/admin/generate" && req.method==="POST"){
    let b; try{b=await req.json();}catch(e){return err("Bad request",400);}
    if(b.admin_key !== env.ADMIN_KEY) return err("无权限",403);
    const CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789"; let code="";
    const rnd=new Uint8Array(8); crypto.getRandomValues(rnd);
    for(let i=0;i<8;i++) code+=CHARS[rnd[i]%CHARS.length];
    const quota = parseInt(b.quota)||500;
    await env.CODES.put("code:"+code, JSON.stringify({quota:quota,used:0,created:Date.now()}));
    return new Response(JSON.stringify({code:code,quota:quota}),{headers:{...cors(),"Content-Type":"application/json"}});
  }

  // --- 公式 OCR（Mathpix）：拍照识别数学公式 ---
  if(url.pathname==="/ocr" && req.method==="POST"){
    const oip = req.headers.get("CF-Connecting-IP")||"unknown";
    const oday = new Date().toISOString().slice(0,10);
    const ok = "ocr:"+oip+":"+oday;
    const on = parseInt(await env.CODES.get(ok)||"0");
    if(on>=20) return err("今日识别次数已用完",402);
    await env.CODES.put(ok, String(on+1), {expirationTtl:172800});
    let ob;
    try{ob=await req.json();}catch(e){return err("Bad request",400);}
    if(!ob.image) return err("Bad request",400);
    const mr = await fetch("https://api.mathpix.com/v3/text",{
      method:"POST",
      headers:{"app_id":env.MATHPIX_APP_ID,"app_key":env.MATHPIX_APP_KEY,"Content-Type":"application/json"},
      body: JSON.stringify({src:ob.image, formats:["text","latex_styled"], data_options:{include_latex:true}})});
    const md = await mr.json();
    return new Response(JSON.stringify({text:md.text||"", latex:md.latex_styled||""}),
      {headers:{...cors(),"Content-Type":"application/json"}});
  }

  if(req.method!=="POST" || url.pathname!=="/") return new Response("Not found",{status:404});

  // --- 卡密 / 免费额度校验 ---
  const code = req.headers.get("X-Access-Code")||"";
  const ip = req.headers.get("CF-Connecting-IP")||"unknown";
  const today = new Date().toISOString().slice(0,10);
  if(code){
    const rec = await env.CODES.get("code:"+code,{type:"json"});
    if(!rec) return err("卡密无效，请检查后重试",403);
    if(rec.used>=rec.quota) return err("该卡密次数已用完",403);
    rec.used++;
    await env.CODES.put("code:"+code, JSON.stringify(rec));
  }else{
    const k = "free:"+ip+":"+today;
    const n = parseInt(await env.CODES.get(k)||"0");
    if(n>=FREE_DAILY) return err("今日免费次数已用完，输入卡密可继续使用",402);
    await env.CODES.put(k, String(n+1), {expirationTtl:172800});
  }

  // --- 转发 DeepSeek（锁定模型与上限，防滥用）---
  let parsed;
  try{parsed=await req.json();}catch(e){return err("Bad request",400);}
  if(!parsed.messages||!Array.isArray(parsed.messages)) return err("Bad request",400);
  parsed.model="deepseek-v4-flash";
  parsed.max_tokens=Math.min(parsed.max_tokens||2000,4000);
  const r=await fetch("https://api.deepseek.com/chat/completions",{
    method:"POST",
    headers:{"Content-Type":"application/json","Authorization":"Bearer "+API_KEY},
    body:JSON.stringify(parsed)});
  const data=await r.text();
  return new Response(data,{headers:{...cors(),"Content-Type":"application/json"}});
}};

第五步：连上网站

  1. Worker 部署成功后得到地址，如 https://xskai-api.xxx.workers.dev。
  2. 打开本站 config.js，把 SITE_WORKER 换成这个地址——两个工具自动生效。
  3. 用 admin.html（不挂导航，直接访问）生成第一张测试卡密，自己试一遍流程。

第六步：闲鱼上架卖卡密

  1. 闲鱼发布虚拟商品："XSKAI 学习卡密"，定价参考 定价页。
  2. 有人下单 → 打开 admin.html 生成卡密 → 闲鱼私聊发码。
  3. 卡密在 KV 里可查已用次数，有纠纷时看记录。
成本账：一次调用约 ¥0.01-0.03。卡密 ¥9.9/500次，成本约 ¥5-15，毛利一半以上。免费 5 次/天是引流成本，单个 IP 一天最多烧你 ¥0.15。
